// Isolated point/triangle conformity experiment. No change to the oracle or mesher.
// NTS supplies only the immutable point hierarchy, never geometric verdicts.
sealed class Search
{
    readonly Mesh2Output mesh;
    readonly Pt[] exact;
    readonly int[] used;
    public Search(Mesh2Output mesh)
    {
        if (mesh.ExactVertices != null) throw new ArgumentException("Pilot accepts binary64 output only");
        if (mesh.VX.Concat(mesh.VY).Any(x => !double.IsFinite(x))) throw new ArgumentException("Non-finite vertex");
        this.mesh = mesh;
        exact = mesh.VX.Select((x,i) => Pt.FromDoubles(x,mesh.VY[i])).ToArray();
        used = mesh.Triangles.SelectMany(t => t).Distinct().OrderBy(i => i).ToArray();
        if (mesh.Triangles.Any(t => Orient(t[0],t[1],t[2])<=0)) throw new ArgumentException("Pilot requires positive triangles, as Conformity does after Orientation");
    }

    // The existing oracle's OrientV filter, unchanged; uncertain cases use its exact predicate.
    int Orient(int a, int b, double cx, double cy, Pt c)
    {
        double ax=mesh.VX[a], ay=mesh.VY[a];
        double l=(mesh.VX[b]-ax)*(cy-ay), rr=(mesh.VY[b]-ay)*(cx-ax);
        double det=l-rr, sum=Math.Abs(l)+Math.Abs(rr), bound=1e-15*sum;
        if (sum>1e-200 && sum<1e200 && Math.Abs(det)>bound) return det>0 ? 1 : -1;
        return Exact.Orient(exact[a],exact[b],c);
    }
    int Orient(int a,int b,int c) => Orient(a,b,mesh.VX[c],mesh.VY[c],exact[c]);
    Envelope TriangleBox(int ti)
    {
        var t=mesh.Triangles[ti];
        return new Envelope(t.Min(i=>mesh.VX[i]),t.Max(i=>mesh.VX[i]),t.Min(i=>mesh.VY[i]),t.Max(i=>mesh.VY[i]));
    }

    sealed class Node
    {
        public Envelope Box;
        public Node[] Children;
        public int Item;
        public Pt[] Corners;
    }
    static Node Wrap(NetTopologySuite.Geometries.IBoundable<Envelope,int> source)
    {
        var e=source.Bounds;
        var node=new Node { Box=e };
        if (source is AbstractNode<Envelope,int> branch) {
            node.Children=branch.ChildBoundables.Select(Wrap).ToArray();
            node.Corners=new[] { Pt.FromDoubles(e.MinX,e.MinY),Pt.FromDoubles(e.MaxX,e.MinY),Pt.FromDoubles(e.MinX,e.MaxY),Pt.FromDoubles(e.MaxX,e.MaxY) };
        } else node.Item=source.Item;
        return node;
    }
    bool Outside(int[] t,Node node)
    {
        for (int edge=0;edge<3;edge++) {
            int a=t[edge], b=t[(edge+1)%3];
            // orient(a,b,p) is affine in p: maximize it over the closed box.
            // Its x coefficient is -(by-ay), its y coefficient is bx-ax.
            bool right=mesh.VY[b]<=mesh.VY[a], top=mesh.VX[b]>=mesh.VX[a];
            int corner=(right?1:0)+(top?2:0);
            double x=right?node.Box.MaxX:node.Box.MinX, y=top?node.Box.MaxY:node.Box.MinY;
            // Strictly outside only: points on edges must still be reported.
            if (Orient(a,b,x,y,node.Corners[corner])<0) return true;
        }
        return false;
    }
    sealed class Visitor : IItemVisitor<int>
    {
        public Action<int> Visit;
        public void VisitItem(int item) => Visit(item);
    }
    static int[] Ranks(double[] values)
    {
        var order=Enumerable.Range(0,values.Length).OrderBy(i => values[i]).ToArray();
        var ranks=new int[values.Length]; int rank=0;
        for (int i=0;i<order.Length;i++) {
            if (i>0 && values[order[i]]!=values[order[i-1]]) rank++;
            ranks[order[i]]=rank;
        }
        return ranks;
    }
    public Result Run(string mode,int queryCount=0)
    {
        if (mode!="boxes" && mode!="nts" && mode!="pruned") throw new ArgumentException("Unknown mode");
        int count=queryCount<=0 ? mesh.Triangles.Length : Math.Min(queryCount,mesh.Triangles.Length);
        var queries=Enumerable.Range(0,count).Select(i => (int)((long)i*mesh.Triangles.Length/count)).ToArray();
        var hits=new List<(int,int)>(); long candidates=0,nodes=0,pruned=0;
        void Check(int ti,int id) {
            var t=mesh.Triangles[ti];
            if (id==t[0] || id==t[1] || id==t[2]) return;
            candidates++;
            if (Orient(t[0],t[1],id)>=0 && Orient(t[1],t[2],id)>=0 && Orient(t[2],t[0],id)>=0) hits.Add((ti,id));
        }
        using var process=Process.GetCurrentProcess();
        var cpuStart=process.TotalProcessorTime;
        var watch=Stopwatch.StartNew();
        if (mode=="boxes") {
            var rx=Ranks(mesh.VX); var ry=Ranks(mesh.VY);
            var tb=queries.Select(i => mesh.Triangles[i]).Select(t => new[] { t.Min(i=>rx[i]),t.Max(i=>rx[i]),t.Min(i=>ry[i]),t.Max(i=>ry[i]) }).ToArray();
            var vb=used.Select(i => new[] { rx[i],rx[i],ry[i],ry[i] }).ToArray();
            BoxPairs.Overlaps(tb,vb,false,(ti,vi) => Check(queries[ti],used[vi]));
        } else {
            var tree=new STRtree<int>();
            foreach (int id in used) tree.Insert(new Envelope(mesh.VX[id],mesh.VX[id],mesh.VY[id],mesh.VY[id]),id);
            tree.Build();
            if (mode=="nts") {
                var visitor=new Visitor();
                foreach (int ti in queries) {
                    visitor.Visit=id => Check(ti,id);
                    tree.Query(TriangleBox(ti),visitor);
                }
            } else {
                var root=Wrap(tree.Root);
                void Walk(Node node,int ti,Envelope triangleBox) {
                    nodes++;
                    if (!node.Box.Intersects(triangleBox)) return;
                    if (node.Children==null) { Check(ti,node.Item); return; }
                    if (Outside(mesh.Triangles[ti],node)) { pruned++; return; }
                    foreach (var child in node.Children) Walk(child,ti,triangleBox);
                }
                foreach (int ti in queries) Walk(root,ti,TriangleBox(ti));
            }
        }
        watch.Stop();
        process.Refresh();
        double cpu=(process.TotalProcessorTime-cpuStart).TotalSeconds;
        hits.Sort();
        string canonical=string.Join("\n",hits.Select(p=>p.Item1+" "+p.Item2));
        string hash=Convert.ToHexString(SHA256.HashData(Encoding.UTF8.GetBytes(canonical))).ToLowerInvariant();
        return new Result(mode,count,watch.Elapsed.TotalSeconds,cpu,candidates,nodes,pruned,hits.Count,hash);
    }
    public Result Exhaustive()
    {
        var hits=new List<(int,int)>();
        for (int ti=0;ti<mesh.Triangles.Length;ti++) foreach (int id in used) {
            var t=mesh.Triangles[ti];
            if (t.Contains(id)) continue;
            // Independent exact leaf test: no floating filter or spatial index.
            if (Exact.Orient(exact[t[0]],exact[t[1]],exact[id])>=0 && Exact.Orient(exact[t[1]],exact[t[2]],exact[id])>=0 && Exact.Orient(exact[t[2]],exact[t[0]],exact[id])>=0) hits.Add((ti,id));
        }
        hits.Sort();
        string canonical=string.Join("\n",hits.Select(p=>p.Item1+" "+p.Item2));
        return new Result("exhaustive",mesh.Triangles.Length,0,0,0,0,0,hits.Count,Convert.ToHexString(SHA256.HashData(Encoding.UTF8.GetBytes(canonical))).ToLowerInvariant());
    }
}
record Result(string Mode,int Queries,double Seconds,double CpuSeconds,long Candidates,long Nodes,long Pruned,int Hits,string Fingerprint);

static class Program
{
    static Mesh2Output Make(double[][] points,int[][] triangles) => new Mesh2Output { VX=points.Select(p=>p[0]).ToArray(),VY=points.Select(p=>p[1]).ToArray(),Triangles=triangles };
    static void SelfTest()
    {
        var cases=new List<Mesh2Output> {
            Make(new[]{new[]{0d,0},new[]{4d,0},new[]{0d,4},new[]{1d,1},new[]{2d,1},new[]{1d,2}},new[]{new[]{0,1,2},new[]{3,4,5}}),
            Make(new[]{new[]{0d,0},new[]{4d,0},new[]{0d,4},new[]{2d,0},new[]{2d,-1},new[]{3d,-1}},new[]{new[]{0,1,2},new[]{3,4,5}}),
            Make(new[]{new[]{0d,0},new[]{4d,0},new[]{0d,4},new[]{0d,0},new[]{-2d,0},new[]{0d,-2}},new[]{new[]{0,1,2},new[]{3,4,5}})
        };
        var random=new Random(117);
        var boundaryPoints=new List<double[]> { new[]{0d,0},new[]{100d,0},new[]{0d,100} };
        var boundaryTriangles=new List<int[]> { new[]{0,1,2} };
        for (int i=0;i<12;i++) {
            int start=boundaryPoints.Count; double x=2+2*i;
            boundaryPoints.AddRange(new[]{new[]{x,0d},new[]{x,-1d},new[]{x+.5,-1d}});
            boundaryTriangles.Add(new[]{start,start+1,start+2});
            start=boundaryPoints.Count; x=60+.2*i;
            boundaryPoints.AddRange(new[]{new[]{x,60d},new[]{x+.05,60d},new[]{x,60.05}});
            boundaryTriangles.Add(new[]{start,start+1,start+2});
        }
        var multiTreeBoundary=Make(boundaryPoints.ToArray(),boundaryTriangles.ToArray());
        cases.Add(multiTreeBoundary);
        for (int sample=0;sample<100;sample++) {
            var points=Enumerable.Range(0,20).Select(_=>new[]{(double)random.Next(-10,11),(double)random.Next(-10,11)}).ToArray();
            var triangles=new List<int[]>();
            for (int attempt=0;attempt<60 && triangles.Count<20;attempt++) {
                int a=random.Next(20),b=random.Next(20),c=random.Next(20);
                int sign=Exact.Orient(Pt.FromDoubles(points[a][0],points[a][1]),Pt.FromDoubles(points[b][0],points[b][1]),Pt.FromDoubles(points[c][0],points[c][1]));
                if (sign!=0) triangles.Add(sign>0?new[]{a,b,c}:new[]{a,c,b});
            }
            cases.Add(Make(points,triangles.ToArray()));
        }
        int checkedCases=0;
        foreach (var original in cases) {
          string originalFingerprint=new Search(original).Exhaustive().Fingerprint;
          foreach (int exponent in new[]{-500,0,500}) {
            double scale=Math.ScaleB(1,exponent);
            var mesh=new Mesh2Output { VX=original.VX.Select(x=>x*scale).ToArray(),VY=original.VY.Select(y=>y*scale).ToArray(),Triangles=original.Triangles };
            var search=new Search(mesh); var expected=search.Exhaustive();
            if (expected.Fingerprint!=originalFingerprint) throw new Exception("Power-of-two invariance differs");
            foreach (string mode in new[]{"boxes","nts","pruned"}) {
                var actual=search.Run(mode);
                if (actual.Fingerprint!=expected.Fingerprint || actual.Hits!=expected.Hits) throw new Exception("Search differs from exact exhaustive check: "+mode);
                if (ReferenceEquals(original,multiTreeBoundary) && mode=="pruned" && actual.Pruned==0) throw new Exception("Targeted case must prune an actual subtree");
                if (search.Run(mode).Fingerprint!=actual.Fingerprint) throw new Exception("Reuse differs");
            }
            Parallel.For(0,3,_=> { if (search.Run("pruned").Fingerprint!=expected.Fingerprint) throw new Exception("Parallel differs"); });
            checkedCases++;
          }
        }
        Console.WriteLine("SELFTEST passed: "+checkedCases+" cases; exact exhaustive equivalence, boundary/nested/coincident cases, power-of-two scales, reuse and parallel determinism.");
    }
    static int Main(string[] args)
    {
        if (args.Length==1 && args[0]=="selftest") { SelfTest(); return 0; }
        if (args.Length<2 || args.Length>3) { Console.Error.WriteLine("OracleIndexPilot <boxes|nts|pruned> <mesh2-output> [query-count; default all]"); return 2; }
        var mesh=TextFormat.ReadOutput(File.ReadAllText(args[1]),out _);
        var search=new Search(mesh);
        Console.WriteLine(JsonSerializer.Serialize(new { phase="start",mode=args[0],vertices=mesh.VertexCount,triangles=mesh.Triangles.Length }));
        Console.WriteLine(JsonSerializer.Serialize(search.Run(args[0],args.Length==3?int.Parse(args[2]):0)));
        return 0;
    }
}
