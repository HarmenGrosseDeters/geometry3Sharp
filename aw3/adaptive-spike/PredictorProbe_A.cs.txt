// Adaptive alpha spike, step A (measurement only, not library code): a cavity-aware level predictor computed from the
// input alone, against the oracle levels of the upper-bound study (aw3/adaptive on runs/adaptive-alpha-study).
// Oracle: W0 at alpha0 = diag/300, offset = diag/3000, W_L at 2^L alpha0; level(t) of a W0 triangle = largest L with
// dev_l(t) <= tau for l <= L (both ways, as in the study).
// Predictor: input samples (spacing alpha0/2); per sample and side the shrinking ball (Ma, Bae, Choi, Kerber, Rousseau 2012:
// the largest empty ball tangent at the sample on that side, against the input); a side counts if its ball center is outside
// the input (fast winding number < 0.5). The fine wrap may reach the sample if r + offset >= klo alpha0; then the level is the
// largest L with r + offset >= khi alpha_L (coarse wraps surely reach it too); otherwise unconstrained. Per W0 triangle the
// predicted level is the minimum over samples within offset + alpha0 of its centroid. Graded as in the study (g = 0.5).
using System; using System.Collections.Generic; using System.Diagnostics; using System.Linq; using System.Threading.Tasks;
using g3; using gs;
static class P {
  static DMesh3 Wrap(DMesh3 input, double alpha, double offset, out long ms) {
    var sw = Stopwatch.StartNew(); var w = new MeshAlphaWrap(input) { Alpha = alpha, Offset = offset }; w.Compute(); ms = sw.ElapsedMilliseconds; return w.Result; }
  static IEnumerable<Vector3d> Samples(DMesh3 m, int t) {
    Vector3d a = Vector3d.Zero, b = Vector3d.Zero, c = Vector3d.Zero; m.GetTriVertices(t, ref a, ref b, ref c);
    yield return (a + b + c) / 3; yield return 0.5 * (a + b); yield return 0.5 * (b + c); yield return 0.5 * (c + a); }
  static double Dist(DMeshAABBTree3 tree, DMesh3 m, Vector3d p) { int t = tree.FindNearestTriangle(p); if (t < 0) return double.MaxValue; return Math.Sqrt(MeshQueries.TriangleDistance(m, t, p).DistanceSquared); }
  // shrinking ball at p with unit normal n, radius cap rmax; returns radius and center
  static double Shrink(DMeshAABBTree3 tree, DMesh3 m, Vector3d p, Vector3d n, double rmax, double eps, out Vector3d center) {
    double r = rmax; center = p + r * n;
    for (int it = 0; it < 40; ++it) {
      int t = tree.FindNearestTriangle(center); if (t < 0) break;
      var q = MeshQueries.TriangleDistance(m, t, center); var qp = q.TriangleClosest;
      double d = Math.Sqrt(q.DistanceSquared);
      if (d >= r - eps) break;
      Vector3d v = qp - p; double den = 2 * v.Dot(n);
      if (v.Length < eps || den <= 0) break;   // the nearest point is p itself (or behind it): converged
      double rn = v.LengthSquared / den; if (rn >= r) break; r = rn; center = p + r * n;
      if (r < eps) break;
    }
    return r;
  }
  static int[] Grade(DMesh3 W0, int[] lev, double alpha0, int levels, double g, Vector3d[] cen) {
    int n = W0.MaxTriangleID; var a = new double[n]; var pq = new SortedSet<(double, int)>();
    foreach (int t in W0.TriangleIndices()) { a[t] = alpha0 * (1 << lev[t]); pq.Add((a[t], t)); }
    while (pq.Count > 0) { var (av, t) = pq.Min; pq.Remove(pq.Min); if (av > a[t]) continue; Index3i nb = W0.GetTriNeighbourTris(t);
      for (int j = 0; j < 3; ++j) { int u = nb[j]; if (u < 0) continue; double cand = av + g * cen[t].Distance(cen[u]); if (cand < a[u]) { pq.Remove((a[u], u)); a[u] = cand; pq.Add((cand, u)); } } }
    var o = new int[n]; foreach (int t in W0.TriangleIndices()) o[t] = Math.Max(0, Math.Min(levels, (int)Math.Floor(Math.Log(a[t] / alpha0, 2) + 1e-9))); return o; }
  static void Main(string[] args) {
    string mesh = args[0]; int levels = 3;
    DMesh3 input = StandardMeshReader.ReadMesh(mesh); double diag = input.CachedBounds.DiagonalLength;
    double alpha0 = diag / 300, offset = diag / 3000;
    var W = new List<DMesh3>(); var ms = new List<long>();
    for (int L = 0; L <= levels; ++L) { W.Add(Wrap(input, alpha0 * (1 << L), offset, out long t)); ms.Add(t); }
    DMesh3 W0 = W[0]; int n = W0.MaxTriangleID; var tree0 = new DMeshAABBTree3(W0, true); var tris = W0.TriangleIndices().ToArray();
    var dev = new double[levels + 1][];
    for (int L = 1; L <= levels; ++L) {
      var d = new double[n]; var treeL = new DMeshAABBTree3(W[L], true);
      Parallel.ForEach(tris, t => { double mm = 0; foreach (var p in Samples(W0, t)) mm = Math.Max(mm, Dist(treeL, W[L], p)); d[t] = mm; });
      var lk = new object();
      Parallel.ForEach(W[L].TriangleIndices().ToArray(), s => { foreach (var p in Samples(W[L], s)) { int t = tree0.FindNearestTriangle(p); if (t < 0) continue;
        double dd = Math.Sqrt(MeshQueries.TriangleDistance(W0, t, p).DistanceSquared); lock (lk) { if (dd > d[t]) d[t] = dd; } } });
      dev[L] = d;
    }
    var rho = new double[levels + 1]; for (int L = 0; L <= levels; ++L) { double ar = 0; foreach (int t in W[L].TriangleIndices()) ar += W[L].GetTriArea(t); rho[L] = W[L].TriangleCount / ar; }
    var cen = new Vector3d[n]; var area = new double[n]; double total = 0;
    foreach (int t in tris) { cen[t] = W0.GetTriCentroid(t); area[t] = W0.GetTriArea(t); total += area[t]; }
    // predictor (timed: the part a real implementation would run before the wrap)
    var swp = Stopwatch.StartNew();
    var inTree = new DMeshAABBTree3(input, true); inTree.FastWindingNumber(Vector3d.Zero);
    double sp = alpha0 / 2, eps = 1e-9 * diag; var rnd = new Random(1);
    var pts = new List<(Vector3d p, Vector3d nn)>();
    foreach (int t in input.TriangleIndices()) {
      Vector3d a = Vector3d.Zero, b = Vector3d.Zero, c = Vector3d.Zero; input.GetTriVertices(t, ref a, ref b, ref c);
      Vector3d nn = (b - a).Cross(c - a); double ar2 = nn.Length; if (!(ar2 > 0)) continue; nn /= ar2;
      int k = Math.Max(1, (int)Math.Ceiling(0.5 * ar2 / (sp * sp))); k = Math.Min(k, 20000);
      for (int j = 0; j < k; ++j) { double u = rnd.NextDouble(), v = rnd.NextDouble(); if (u + v > 1) { u = 1 - u; v = 1 - v; } pts.Add((a + u * (b - a) + v * (c - a), nn)); }
    }
    double rmax = 2 * alpha0 * (1 << levels);
    var rad = new double[pts.Count, 2]; var ext = new bool[pts.Count, 2];
    Parallel.For(0, pts.Count, i => { for (int s = 0; s < 2; ++s) { var nn = s == 0 ? pts[i].nn : -pts[i].nn;
      double r = Shrink(inTree, input, pts[i].p + 1e-7 * diag * nn, nn, rmax, eps, out Vector3d cc);
      rad[i, s] = r; ext[i, s] = inTree.FastWindingNumber(cc) < 0.5; } });
    long predMs = swp.ElapsedMilliseconds;
    // spatial hash of samples
    double cell = offset + alpha0; var grid = new Dictionary<(int, int, int), List<int>>();
    (int, int, int) key(Vector3d p) => ((int)Math.Floor(p.x / cell), (int)Math.Floor(p.y / cell), (int)Math.Floor(p.z / cell));
    for (int i = 0; i < pts.Count; ++i) { var kk = key(pts[i].p); if (!grid.TryGetValue(kk, out var l)) grid[kk] = l = new List<int>(); l.Add(i); }
    var rows = new List<string>();
    foreach (double tauF in new[] { 1.0, 2.0 }) {
      double tau = tauF * offset; var lev = new int[n];
      foreach (int t in tris) { int L = 0; while (L < levels && dev[L + 1][t] <= tau) L++; lev[t] = L; }
      var gl = Grade(W0, lev, alpha0, levels, 0.5, cen);
      double estO = 0; foreach (int t in tris) estO += area[t] * rho[gl[t]];
      rows.Add($"\"tau{tauF}_oracle_g\":{estO:F0}");
      foreach (var (klo, khi, useExt) in new[] { (0.7, 1.3, true), (0.5, 1.5, true), (0.7, 1.3, false), (0.5, 2.0, true) }) {
        // sample level
        var sl = new int[pts.Count];
        for (int i = 0; i < pts.Count; ++i) { int Lmin = levels; for (int s = 0; s < 2; ++s) { if (useExt && !ext[i, s]) continue; double rr = rad[i, s] + offset;
          if (rr < klo * alpha0) continue; int L = 0; while (L < levels && rr >= khi * alpha0 * (1 << (L + 1))) L++; Lmin = Math.Min(Lmin, L); } sl[i] = Lmin; }
        var pl = new int[n];
        Parallel.ForEach(tris, t => { var kc = key(cen[t]); int m = levels; double r2 = cell * cell;
          for (int dx = -1; dx <= 1; ++dx) for (int dy = -1; dy <= 1; ++dy) for (int dz = -1; dz <= 1; ++dz)
            if (grid.TryGetValue((kc.Item1 + dx, kc.Item2 + dy, kc.Item3 + dz), out var l)) foreach (int i in l) if (sl[i] < m && pts[i].p.DistanceSquared(cen[t]) <= r2) m = sl[i];
          pl[t] = m; });
        var pg = Grade(W0, pl, alpha0, levels, 0.5, cen);
        double fa = 0, fag = 0, worst = 0, est = 0;
        foreach (int t in tris) { if (pl[t] > lev[t]) fa += area[t]; if (pg[t] > lev[t]) { fag += area[t]; worst = Math.Max(worst, dev[pg[t]][t] / offset); } est += area[t] * rho[pg[t]]; }
        string nm = $"tau{tauF}_k{klo}_{khi}{(useExt ? "" : "_noext")}";
        rows.Add($"\"{nm}_false\":{fa / total:F4},\"{nm}_false_g\":{fag / total:F4},\"{nm}_worst\":{worst:F1},\"{nm}_est\":{est:F0}");
      }
    }
    Console.WriteLine("{\"mesh\":\"" + System.IO.Path.GetFileNameWithoutExtension(mesh) + "\",\"raw\":[" + string.Join(",", W.Select(w => w.TriangleCount)) +
      "],\"ms\":[" + string.Join(",", ms) + "],\"pred_ms\":" + predMs + ",\"samples\":" + pts.Count + "," + string.Join(",", rows) + "}");
  }
}
